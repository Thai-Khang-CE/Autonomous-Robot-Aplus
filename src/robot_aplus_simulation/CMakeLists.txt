cmake_minimum_required(VERSION 3.16)
project(robot_aplus_simulation)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(obstacle_patrol SHARED src/obstacle_patrol.cpp)
target_compile_features(obstacle_patrol PUBLIC cxx_std_17)
target_link_libraries(obstacle_patrol gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS obstacle_patrol LIBRARY DESTINATION lib)
install(DIRECTORY launch worlds urdf config DESTINATION share/${PROJECT_NAME})
ament_package()
